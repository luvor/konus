cmake_minimum_required(VERSION 3.21)
set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Qt 6.8 minimum")
project(Konus VERSION 0.11.0 DESCRIPTION "GPL-3.0-or-later desktop video player" LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2 Multimedia Svg)
if(APPLE)
  set(VLC_SDK_DIR "/Applications/VLC.app/Contents/MacOS" CACHE PATH "libVLC SDK root")
  find_path(VLC_INCLUDE_DIR vlc/vlc.h PATHS "${VLC_SDK_DIR}/include" REQUIRED)
  find_library(VLC_LIBRARY vlc PATHS "${VLC_SDK_DIR}/lib" REQUIRED)
elseif(WIN32)
  set(VLC_SDK_DIR "${CMAKE_SOURCE_DIR}/sdk" CACHE PATH "libVLC SDK root")
  find_path(VLC_INCLUDE_DIR vlc/vlc.h PATHS "${VLC_SDK_DIR}/include" REQUIRED)
  find_library(VLC_LIBRARY libvlc PATHS "${VLC_SDK_DIR}/lib" REQUIRED)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(VLC REQUIRED IMPORTED_TARGET libvlc)
endif()
qt_standard_project_setup(REQUIRES 6.5)
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_executable(konus src/main.cpp src/player.cpp src/player.h)
if(WIN32)
  enable_language(RC)
  configure_file(cmake/konus.rc.in "${CMAKE_CURRENT_BINARY_DIR}/konus.rc" @ONLY)
  target_sources(konus PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/konus.rc")
  set_target_properties(konus PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
target_compile_definitions(konus PRIVATE KONUS_VERSION="${PROJECT_VERSION}")
qt_add_resources(konus konus_app_icon PREFIX "/konus/assets/app" BASE "${CMAKE_CURRENT_SOURCE_DIR}/assets/app" FILES assets/app/konus.svg)
qt_add_qml_module(konus URI Konus VERSION 1.0 QML_FILES qml/Main.qml qml/Theme.qml qml/Icon.qml)
file(GLOB KONUS_ICONS CONFIGURE_DEPENDS assets/icons/*.svg)
qt_add_resources(konus konus_icons PREFIX "/konus/assets/icons" BASE "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons" FILES ${KONUS_ICONS})
file(GLOB KONUS_FONTS CONFIGURE_DEPENDS assets/fonts/*.ttf assets/fonts/*.otf assets/fonts/OFL.txt)
qt_add_resources(konus konus_fonts PREFIX "/konus/assets/fonts" BASE "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts" FILES ${KONUS_FONTS})
target_link_libraries(konus PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Svg)
if(WIN32 OR APPLE)
  target_include_directories(konus PRIVATE "${VLC_INCLUDE_DIR}")
  target_link_libraries(konus PRIVATE "${VLC_LIBRARY}")
else()
  target_link_libraries(konus PRIVATE PkgConfig::VLC)
endif()
if(APPLE)
  target_sources(konus PRIVATE assets/app/konus.icns)
  set_source_files_properties(assets/app/konus.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(konus PROPERTIES
    MACOSX_BUNDLE_ICON_FILE "konus.icns"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}")
  set_target_properties(konus PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist.in" MACOSX_BUNDLE_GUI_IDENTIFIER "org.konus.player" MACOSX_BUNDLE_BUNDLE_NAME "Konus" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks;${VLC_SDK_DIR}/lib")
endif()

option(KONUS_BUILD_FRAME_CHECK "Build the dropped-picture regression check" OFF)
if(KONUS_BUILD_FRAME_CHECK)
  qt_add_executable(konus-frame-check tests/player_frame_check.cpp src/player.cpp src/player.h)
  target_link_libraries(konus-frame-check PRIVATE Qt6::Multimedia)
  if(WIN32 OR APPLE)
    target_include_directories(konus-frame-check PRIVATE "${VLC_INCLUDE_DIR}")
    target_link_libraries(konus-frame-check PRIVATE "${VLC_LIBRARY}")
  else()
    target_link_libraries(konus-frame-check PRIVATE PkgConfig::VLC)
  endif()
  if(APPLE)
    target_link_options(konus-frame-check PRIVATE "-Wl,-rpath,${VLC_SDK_DIR}/lib")
  endif()
endif()
