cmake_minimum_required(VERSION 3.21)
project(Konus VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Quick QuickControls2 Multimedia Svg)
if(APPLE)
  set(VLC_SDK_DIR "/Applications/VLC.app/Contents/MacOS" CACHE PATH "libVLC SDK root")
  find_path(VLC_INCLUDE_DIR vlc/vlc.h PATHS "${VLC_SDK_DIR}/include" REQUIRED)
  find_library(VLC_LIBRARY vlc PATHS "${VLC_SDK_DIR}/lib" REQUIRED)
elseif(WIN32)
  set(VLC_SDK_DIR "${CMAKE_SOURCE_DIR}/sdk" CACHE PATH "libVLC SDK root")
  find_path(VLC_INCLUDE_DIR vlc/vlc.h PATHS "${VLC_SDK_DIR}/include" REQUIRED)
  find_library(VLC_LIBRARY libvlc PATHS "${VLC_SDK_DIR}/lib" REQUIRED)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(VLC REQUIRED IMPORTED_TARGET libvlc)
endif()
qt_standard_project_setup(REQUIRES 6.4)
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_executable(konus src/main.cpp src/player.cpp src/player.h)
qt_add_qml_module(konus URI Konus VERSION 1.0 QML_FILES qml/Main.qml qml/Theme.qml)
qt_add_resources(konus konus_icons PREFIX "/konus" FILES assets/icons/cone.svg)
target_link_libraries(konus PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Svg)
if(WIN32 OR APPLE)
  target_include_directories(konus PRIVATE "${VLC_INCLUDE_DIR}")
  target_link_libraries(konus PRIVATE "${VLC_LIBRARY}")
else()
  target_link_libraries(konus PRIVATE PkgConfig::VLC)
endif()
if(APPLE)
  set_target_properties(konus PROPERTIES MACOSX_BUNDLE TRUE)
  target_link_options(konus PRIVATE "-Wl,-rpath,${VLC_SDK_DIR}/lib")
endif()
